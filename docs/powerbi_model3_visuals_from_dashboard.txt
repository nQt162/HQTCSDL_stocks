DAX measure goi y
-----------------
Tao trong Modeling -> New measure.

1. Accuracy

M3 Accuracy % =
DIVIDE(
    CALCULATE(
        MAX(mart_model3_metrics[metric_value]),
        mart_model3_metrics[metric_name] = "Accuracy"
    ),
    100
)

Format: Percentage.


2. Macro F1

M3 Macro F1 =
CALCULATE(
    MAX(mart_model3_metrics[metric_value]),
    mart_model3_metrics[metric_name] = "Macro_F1"
)

Format: Decimal, 4 chu so.


3. BUY Count Adjusted

M3 Adjusted BUY Count =
CALCULATE(
    COUNTROWS(mart_model3_trading_signals),
    mart_model3_trading_signals[adjusted_signal] = "BUY"
)


4. SELL Count Adjusted

M3 Adjusted SELL Count =
CALCULATE(
    COUNTROWS(mart_model3_trading_signals),
    mart_model3_trading_signals[adjusted_signal] = "SELL"
)


5. HOLD Count Adjusted

M3 Adjusted HOLD Count =
CALCULATE(
    COUNTROWS(mart_model3_trading_signals),
    mart_model3_trading_signals[adjusted_signal] = "HOLD"
)


6. Net Return moi nhat

M3 Net Return % =
VAR LastDate =
    MAX(mart_model3_backtest_daily[trading_date])
RETURN
    CALCULATE(
        MAX(mart_model3_backtest_daily[cumulative_return_net]),
        mart_model3_backtest_daily[trading_date] = LastDate
    )

Format: Percentage.


7. Net Sharpe tot nhat

M3 Best Net Sharpe =
MAX(mart_model3_backtest_sweep[Sharpe_Ratio_Net])


8. BUY Precision tu adjusted signal
Luu y: day la cach tinh tu bang prediction, co the khac mot chut voi HTML neu dashboard co logic rieng.

M3 BUY Precision % =
VAR BuyRows =
    FILTER(
        mart_model3_trading_signals,
        mart_model3_trading_signals[adjusted_signal] = "BUY"
    )
VAR CorrectBuy =
    FILTER(
        BuyRows,
        mart_model3_trading_signals[target_signal] = "BUY"
    )
RETURN
    DIVIDE(COUNTROWS(CorrectBuy), COUNTROWS(BuyRows))

Format: Percentage.


9. SELL Precision tu adjusted signal

M3 SELL Precision % =
VAR SellRows =
    FILTER(
        mart_model3_trading_signals,
        mart_model3_trading_signals[adjusted_signal] = "SELL"
    )
VAR CorrectSell =
    FILTER(
        SellRows,
        mart_model3_trading_signals[target_signal] = "SELL"
    )
RETURN
    DIVIDE(COUNTROWS(CorrectSell), COUNTROWS(SellRows))

Format: Percentage.
